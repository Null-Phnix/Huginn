```python
from enum import Enum

class WaitStrategy(str, Enum):
    """Smart wait strategies for page navigation."""
    TIMEOUT = "timeout"
    SELECTOR = "selector"
    NETWORK_IDLE = "networkidle"
    DOM_CONTENT_LOADED = "domcontentloaded"


def parse_wait_for(wait_for):
    """Parse the wait_for parameter into a (WaitStrategy, value) tuple.
    
    Accepts:
        int/float — timeout in ms: (TIMEOUT, ms)
        "css.selector" — wait for element: (SELECTOR, "css.selector")  
        "networkidle" — wait for network idle: (NETWORK_IDLE, default_ms)
        "domcontentloaded" — wait for DOM ready: (DOM_CONTENT_LOADED, 0)
    """
    if isinstance(wait_for, (int, float)):
        return (WaitStrategy.TIMEOUT, int(wait_for))
    if isinstance(wait_for, str):
        if wait_for in ("networkidle", "network_idle", "network-idle"):
            return (WaitStrategy.NETWORK_IDLE, 5000)
        if wait_for == "domcontentloaded":
            return (WaitStrategy.DOM_CONTENT_LOADED, 0)
        # Otherwise treat as CSS selector
        return (WaitStrategy.SELECTOR, wait_for)
    return (WaitStrategy.TIMEOUT, 3000)
```

```python
async def smart_wait(self, page: Page, strategy: WaitStrategy, value, timeout_ms: int = 10000):
    """Apply intelligent wait strategy to the page.
    
    Args:
        page: Playwright page object
        strategy: WaitStrategy enum value
        value: Timeout in ms (TIMEOUT), selector string (SELECTOR), or unused
        timeout_ms: Max time to wait for selector/networkidle
    """
    if strategy == WaitStrategy.TIMEOUT:
        await asyncio.sleep(value / 1000)
    elif strategy == WaitStrategy.SELECTOR:
        try:
            await page.wait_for_selector(value, timeout=timeout_ms)
        except Exception as e:
            logger.warning(f"Selector wait failed for '{value}': {e}")
    elif strategy == WaitStrategy.NETWORK_IDLE:
        try:
            await page.wait_for_load_state("networkidle", timeout=timeout_ms)
        except Exception:
            pass  # networkidle can timeout on complex pages, that's ok
    elif strategy == WaitStrategy.DOM_CONTENT_LOADED:
        try:
            await page.wait_for_load_state("domcontentloaded")
        except Exception:
            pass  # already loaded
```

```python
import hashlib

def content_hash(text: str) -> str:
    """Hash cleaned text for duplicate detection.
    
    Normalizes whitespace before hashing so superficially different
    content (extra spaces, newlines) hashes the same.
    """
    import re
    # Normalize whitespace: collapse runs of whitespace to single space, strip
    cleaned = re.sub(r'\s+', ' ', text).strip().lower()
    return hashlib.sha256(cleaned.encode('utf-8')).hexdigest()[:16]
```

```python
def classify_error(error: Exception) -> tuple:
    """Classify an error and return (error_type, http_status_code)."""
    import httpx
    if isinstance(error, asyncio.TimeoutError):
        return ("timeout", 408)
    if isinstance(error, (httpx.ConnectError, ConnectionError, ConnectionRefusedError)):
        return ("connection", 502)
    if isinstance(error, httpx.HTTPStatusError):
        code = error.response.status_code
        if code == 429:
            return ("rate_limited", 429)
        if 400 <= code < 500:
            return ("client_error", code)
        if 500 <= code < 600:
            return ("server_error", 502)
    return ("unknown", 500)
```

Then update the `scrape()` method to wrap navigation in a retry loop:

```python
async def scrape(self, url, ..., max_retries: int = 2) -> ScrapeData:
    """Scrape with automatic retry on transient errors."""
    last_error = None
    for attempt in range(max_retries + 1):
        try:
            # ... existing scrape logic ...
            # return result on success
        except (asyncio.TimeoutError, Exception) as e:
            error_type, status = classify_error(e)
            logger.warning(f"Scrape attempt {attempt+1} failed for {url}: {error_type} ({status})")
            last_error = e
            # Don't retry client errors (4xx) or rate limits (429)
            if error_type == "client_error":
                return ScrapeData(metadata={"url": url, "status_code": status, "error": str(e)})
            if error_type == "rate_limited":
                await asyncio.sleep(min(2 ** attempt, 10))  # backoff: 1s, 2s, 4s, max 10s
                continue
            # Retry on timeout, connection, server errors
            if attempt < max_retries:
                await asyncio.sleep(min(2 ** attempt, 5))  # backoff: 1s, 2s
                continue
            # All retries exhausted
            return ScrapeData(metadata={"url": url, "status_code": status, "error": str(e)})
```

```python
import httpx

class RobotsChecker:
    """Check robots.txt for allowed/disallowed paths."""
    
    def __init__(self, base_url: str):
        self.base_url = base_url.rstrip('/')
        self.rules = None  # None = not fetched yet
    
    async def fetch_rules(self):
        """Fetch and parse robots.txt from the site."""
        try:
            async with httpx.AsyncClient(timeout=10) as client:
                parsed = urlparse(self.base_url)
                robots_url = f"{parsed.scheme}://{parsed.netloc}/robots.txt"
                resp = await client.get(robots_url, follow_redirects=True)
                if resp.status_code == 200:
                    self.rules = self._parse(resp.text)
                else:
                    self.rules = {"disallow": [], "allow": []}  # No robots.txt = allow all
        except Exception:
            self.rules = {"disallow": [], "allow": []}  # On error, allow all
    
    def is_allowed(self, path: str) -> bool:
        """Check if a path is allowed by robots.txt rules."""
        if self.rules is None:
            return True  # Not fetched yet, allow
        # Check disallow rules first
        for pattern in self.rules.get("disallow", []):
            if path.startswith(pattern) or pattern == "/":
                # Check if there's a more specific allow rule
                for allow_pattern in self.rules.get("allow", []):
                    if path.startswith(allow_pattern) and len(allow_pattern) > len(pattern):
                        return True
                return False
        return True
    
    def _parse(self, robots_text: str) -> dict:
        """Simple robots.txt parser for User-agent: *."""
        disallow = []
        allow = []
        in_wildcard = False
        for line in robots_text.splitlines():
            line = line.strip()
            if line.startswith("#") or not line:
                continue
            if line.lower().startswith("user-agent:"):
                agent = line.split(":", 1)[1].strip()
                in_wildcard = agent == "*"
            elif in_wildcard:
                if line.lower().startswith("disallow:"):
                    path = line.split(":", 1)[1].strip()
                    if path:
                        disallow.append(path)
                elif line.lower().startswith("allow:"):
                    path = line.split(":", 1)[1].strip()
                    if path:
                        allow.append(path)
        return {"disallow": disallow, "allow": allow}
```

```python
class ScrollConfig(BaseModel):
    """Configuration for infinite scroll handling."""
    max_scrolls: int = 10
    delay_ms: int = 500
    scroll_to_bottom: bool = True

async def auto_scroll(self, page: Page, config: ScrollConfig = None) -> int:
    """Auto-scroll a page to load lazy/dynamic content.
    
    Returns the number of scrolls performed.
    """
    if config is None:
        config = ScrollConfig()
    
    scroll_count = 0
    last_height = 0
    
    for i in range(config.max_scrolls):
        current_height = await page.evaluate("document.body.scrollHeight")
        if current_height == last_height:
            break  # No more content loading
        
        last_height = current_height
        
        # Scroll to bottom
        await page.evaluate(f"window.scrollTo(0, document.body.scrollHeight)")
        scroll_count += 1
        
        # Wait for new content to load
        await asyncio.sleep(config.delay_ms / 1000)
    
    # Scroll back to top
    if config.scroll_to_bottom:
        await page.evaluate("window.scrollTo(0, 0)")
    
    return scroll_count
```

In `huginn/crawler.py`:

```python
# Pagination URL patterns
_PAGINATION_PATTERNS = [
    re.compile(r'/page/(\d+)'),      # /page/1, /page/2
    re.compile(r'[?&]page=(\d+)'),   # ?page=1, &page=2
    re.compile(r'/p/(\d+)'),         # /p/1
    re.compile(r'[?&]p=(\d+)'),       # ?p=1
]

def detect_pagination(current_url: str, links: List[str]) -> List[str]:
    """Detect pagination links from a list of discovered links.
    
    Returns links to pages that come AFTER the current page.
    """
    current_parsed = urlparse(current_url)
    pagination_links = []
    
    for link in links:
        parsed = urlparse(link)
        # Must be same domain
        if parsed.netloc != current_parsed.netloc:
            continue
        
        for pattern in _PAGINATION_PATTERNS:
            match = pattern.search(parsed.path + '?' + parsed.query if parsed.query else parsed.path)
            if match:
                try:
                    page_num = int(match.group(1))
                    # Find current page number
                    current_match = pattern.search(current_parsed.path + '?' + current_parsed.query if current_parsed.query else current_parsed.path)
                    if current_match:
                        current_page = int(current_match.group(1))
                        if page_num == current_page + 1:
                            pagination_links.append(link)
                except (ValueError, IndexError):
                    continue
    
    return pagination_links
```
